## Step 2：实现用户绑定

用户绑定是建立 Agent 与用户的 1:1 关系的关键步骤。用户通过 SMS 验证完成绑定后，你可以使用返回的 User Token 调用 API。

### 绑定流程

```python
import requests
import os

api_key = os.getenv("CLAWDOT_API_KEY")
base_url = "https://api.clawdot.com"

# Step 2a：请求 SMS 验证码
def request_verification_code(phone: str) -> str:
    """
    请求发送 SMS 验证码
    返回: request_id（用于后续验证）
    """
    url = f"{base_url}/api/v1/auth/sms/request"
    payload = {"phone": phone}
    headers = {"Authorization": f"Bearer {api_key}"}

    response = requests.post(url, json=payload, headers=headers)
    response.raise_for_status()

    data = response.json()
    print(f"验证码已发送，{data['expires_in']} 秒内有效")
    return data["request_id"]  # 返回给用户，等待他们输入验证码


# Step 2b：验证码验证，获取 User Token
def verify_and_bind_user(request_id: str, code: str) -> str:
    """
    验证 SMS 验证码，获取 User Token
    返回: user_token（后续 API 调用必需）
    """
    url = f"{base_url}/api/v1/auth/sms/verify"
    payload = {"request_id": request_id, "code": code}
    headers = {"Authorization": f"Bearer {api_key}"}

    response = requests.post(url, json=payload, headers=headers)

    if response.status_code == 400:
        # 验证码错误或已过期
        raise ValueError("验证码错误或已过期，请重新获取")
    elif response.status_code == 401:
        raise ValueError("请求 ID 无效，请重新请求验证码")

    response.raise_for_status()

    data = response.json()
    user_token = data["user_token"]

    # 存储 user_token（用于后续 API 调用）
    # 建议使用安全的本地存储或会话管理
    print(f"用户绑定成功！User Token: {user_token[:20]}...")
    return user_token
```

### 完整绑定 UI 流程

```python
def bind_user_interactive():
    """
    交互式用户绑定（Agent 对话流程）
    """
    print("欢迎使用本地生活 Agent，请先绑定你的账号")
    print("=" * 50)

    # 第一步：获取用户电话号码
    phone = input("请输入你的手机号码（+86 开头）: ").strip()

    if not phone.startswith("+86"):
        phone = "+86" + phone

    # 第二步：请求验证码
    try:
        request_id = request_verification_code(phone)
    except requests.exceptions.RequestException as e:
        print(f"请求失败：{e}")
        return None

    # 第三步：用户输入验证码
    max_attempts = 3
    for attempt in range(max_attempts):
        code = input(f"\n请输入验证码 (剩余 {max_attempts - attempt} 次): ").strip()

        try:
            user_token = verify_and_bind_user(request_id, code)
            print("\n绑定成功！现在你可以开始使用本地生活服务了。")
            return user_token
        except ValueError as e:
            print(f"验证失败：{e}")

    print("\n验证次数过多，请稍后重试或重新获取验证码")
    return None


# 使用示例
user_token = bind_user_interactive()
if user_token:
    # 保存 user_token 供后续使用
    # 例如存储到文件或数据库
    pass
```

User Token 的有效期为 90 天。建议在 Token 即将过期时（剩余 30 天）提示用户重新绑定。
</Info>

def search_shops(query: str, location: str, user_token: str):
    """
    搜索附近的商店
    """
    url = f"{base_url}/api/v1/gateway/shops/search"
    headers = {
        "Authorization": f"Bearer {api_key}",
        "X-User-Token": user_token,
        "Content-Type": "application/json"
    }
    payload = {
        "query": query,
        "location": location,
        "limit": 20
    }

    response = requests.post(url, json=payload, headers=headers)
    response.raise_for_status()

    shops = response.json()["shops"]
    return shops

## Step 4：处理边界情况

健壮的 Agent 应该优雅地处理各种边界情况，提供友好的错误提示。

### 错误处理

```python
def safe_api_call(func, *args, **kwargs):
    """
    包装 API 调用，统一处理错误
    """
    try:
        return func(*args, **kwargs)
    except requests.exceptions.Timeout:
        print("❌ 网络超时，请检查网络连接")
        return None
    except requests.exceptions.ConnectionError:
        print("❌ 无法连接到服务器，请稍后重试")
        return None
    except requests.exceptions.HTTPError as e:
        status_code = e.response.status_code

        if status_code == 400:
            print("❌ 请求参数错误，请检查输入")
        elif status_code == 401:
            print("❌ API Key 或 User Token 无效，请重新绑定")
        elif status_code == 403:
            print("❌ 权限拒绝，请确保账号已激活")
        elif status_code == 429:
            print("❌ 请求过于频繁，请稍后再试")
        elif status_code == 500:
            print("❌ 服务器错误，请稍后重试")
        else:
            print(f"❌ API 错误 ({status_code})：{e}")

        return None
    except Exception as e:
        print(f"❌ 未知错误：{e}")
        return None


# 使用
shops = safe_api_call(search_shops, "外卖", "中关村", user_token)
```

def get_order_with_retry(order_id: str, user_token: str, max_retries: int = 3):
    """
    查询订单，支持重试机制

    for attempt in range(max_retries):

## 完整示例代码

一个完整的本地生活 Agent 示例：

```python
import os
import requests
from typing import List, Dict, Optional

class LocalLifeAgent:
    def __init__(self, api_key: str = None):
        self.api_key = api_key or os.getenv("CLAWDOT_API_KEY")
        self.base_url = "https://api.clawdot.com"
        self.user_token = None
        self.user_phone = None

    def bind_user(self, phone: str) -> bool:
        """用户绑定流程"""
        try:
            # 请求验证码
            request_id = self._request_verification(phone)
            # 输入验证码
            code = input("请输入验证码: ")
            # 验证并获取 token
            self.user_token = self._verify_code(request_id, code)
            self.user_phone = phone
            return True
        except Exception as e:
            print(f"绑定失败：{e}")
            return False

    def search_and_order(self, query: str, location: str) -> Optional[Dict]:
        """搜索商店并下单"""
        if not self.user_token:
            print("请先绑定用户账号")
            return None

        # 搜索商店
        shops = self._search_shops(query, location)
        if not shops:
            print("未找到相关商店")
            return None

        # 展示搜索结果
        for i, shop in enumerate(shops[:5], 1):
            print(f"{i}. {shop['name']} - 评分 {shop['rating']}⭐")

        # 用户选择
        choice = int(input("选择商店 (1-5): ")) - 1
        shop = shops[choice]

        # 获取菜单
        shop_detail = self._get_shop_detail(shop['id'])
        print(f"\n{shop_detail['name']} 菜单：")

        # 构建订单
        items = []
        for category in shop_detail['categories']:
            print(f"\n【{category['name']}】")
            for item in category['items']:
                print(f"  {item['name']} - ¥{item['price']}")

        # 创建订单
        order = self._create_order(shop['id'], items, location)
        return order

    def _search_shops(self, query: str, location: str) -> List[Dict]:
        """搜索商店"""
        url = f"{self.base_url}/api/v1/gateway/shops/search"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token
        }
        response = requests.post(url, json={"query": query, "location": location}, headers=headers)
        return response.json().get("shops", [])

    def _get_shop_detail(self, shop_id: str) -> Dict:
        """获取商店详情"""
        url = f"{self.base_url}/api/v1/gateway/shops/{shop_id}"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token
        }
        response = requests.get(url, headers=headers)
        return response.json()

    def _create_order(self, shop_id: str, items: List[Dict], location: str) -> Dict:
        """创建订单"""
        url = f"{self.base_url}/api/v1/gateway/orders"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token,
            "Content-Type": "application/json"
        }
        payload = {
            "shop_id": shop_id,
            "items": items,
            "delivery_address": location,
            "contact_phone": self.user_phone
        }
        response = requests.post(url, json=payload, headers=headers)
        return response.json()

    def _request_verification(self, phone: str) -> str:
        """请求 SMS 验证码"""
        url = f"{self.base_url}/api/v1/auth/sms/request"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        response = requests.post(url, json={"phone": phone}, headers=headers)
        return response.json()["request_id"]

    def _verify_code(self, request_id: str, code: str) -> str:
        """验证 SMS 验证码"""
        url = f"{self.base_url}/api/v1/auth/sms/verify"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        response = requests.post(
            url,
            json={"request_id": request_id, "code": code},
            headers=headers
        )
        return response.json()["user_token"]


# 使用示例
if __name__ == "__main__":
    agent = LocalLifeAgent()

    # 绑定用户
    if agent.bind_user("+86 138 0000 0000"):
        # 搜索并下单
        order = agent.search_and_order("外卖", "中关村大街1号")
        if order:
            print(f"\n订单已创建: {order['order_id']}")
```
