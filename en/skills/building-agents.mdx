## Step 2: Implement User Binding

User binding is the key step in establishing the 1:1 relationship between Agent and user. After SMS verification, users can use the returned User Token to call APIs.

### Binding Flow

```python
import requests
import os

api_key = os.getenv("CLAWDOT_API_KEY")
base_url = "https://api.clawdot.com"

# Step 2a: Request SMS verification code
def request_verification_code(phone: str) -> str:
    """
    Request SMS verification code
    Return: request_id (for subsequent verification)
    """
    url = f"{base_url}/api/v1/auth/sms/request"
    payload = {"phone": phone}
    headers = {"Authorization": f"Bearer {api_key}"}

    response = requests.post(url, json=payload, headers=headers)
    response.raise_for_status()

    data = response.json()
    print(f"Code sent, valid for {data['expires_in']} seconds")
    return data["request_id"]  # Return to user, wait for code input


# Step 2b: Verify code, get User Token
def verify_and_bind_user(request_id: str, code: str) -> str:
    """
    Verify SMS code, get User Token
    Return: user_token (required for subsequent API calls)
    """
    url = f"{base_url}/api/v1/auth/sms/verify"
    payload = {"request_id": request_id, "code": code}
    headers = {"Authorization": f"Bearer {api_key}"}

    response = requests.post(url, json=payload, headers=headers)

    if response.status_code == 400:
        # Code error or expired
        raise ValueError("Code incorrect or expired, please re-request")
    elif response.status_code == 401:
        raise ValueError("Request ID invalid, please re-request code")

    response.raise_for_status()

    data = response.json()
    user_token = data["user_token"]

    # Store user_token (for subsequent API calls)
    # Recommend using secure local storage or session management
    print(f"User binding successful! User Token: {user_token[:20]}...")
    return user_token
```

### Complete Interactive Binding UI Flow

```python
def bind_user_interactive():
    """
    Interactive user binding (Agent conversation flow)
    """
    print("Welcome to Local Life Agent, please bind your account")
    print("=" * 50)

    # Step 1: Get user phone number
    phone = input("Enter your phone number (+86 format): ").strip()

    if not phone.startswith("+86"):
        phone = "+86" + phone

    # Step 2: Request verification code
    try:
        request_id = request_verification_code(phone)
    except requests.exceptions.RequestException as e:
        print(f"Request failed: {e}")
        return None

    # Step 3: User enters verification code
    max_attempts = 3
    for attempt in range(max_attempts):
        code = input(f"\nEnter verification code ({max_attempts - attempt} attempts left): ").strip()

        try:
            user_token = verify_and_bind_user(request_id, code)
            print("\nBinding successful! You can now use local life services.")
            return user_token
        except ValueError as e:
            print(f"Verification failed: {e}")

    print("\nToo many attempts, please retry later or re-request code")
    return None


# Usage example
user_token = bind_user_interactive()
if user_token:
    # Save user_token for later use
    # e.g. store to file or database
    pass
```

<Info>
User Token validity is 90 days. Recommend prompting user to re-bind when Token is about to expire (30 days remaining).
</Info>

def search_shops(query: str, location: str, user_token: str):
    """
    Search nearby shops
    """
    url = f"{base_url}/api/v1/gateway/shops/search"
    headers = {
        "Authorization": f"Bearer {api_key}",
        "X-User-Token": user_token,
        "Content-Type": "application/json"
    }
    payload = {
        "query": query,
        "location": location,
        "limit": 20
    }

    response = requests.post(url, json=payload, headers=headers)
    response.raise_for_status()

    shops = response.json()["shops"]
    return shops

## Step 4: Handle Edge Cases

Robust Agents gracefully handle edge cases and provide friendly error messages.

### Error Handling

```python
def safe_api_call(func, *args, **kwargs):
    """
    Wrap API call, handle errors uniformly
    """
    try:
        return func(*args, **kwargs)
    except requests.exceptions.Timeout:
        print("❌ Network timeout, check connection")
        return None
    except requests.exceptions.ConnectionError:
        print("❌ Cannot connect to server, retry later")
        return None
    except requests.exceptions.HTTPError as e:
        status_code = e.response.status_code

        if status_code == 400:
            print("❌ Request parameter error, check input")
        elif status_code == 401:
            print("❌ API Key or User Token invalid, please re-bind")
        elif status_code == 403:
            print("❌ Permission denied, ensure account is activated")
        elif status_code == 429:
            print("❌ Request too frequent, retry later")
        elif status_code == 500:
            print("❌ Server error, retry later")
        else:
            print(f"❌ API error ({status_code}): {e}")

        return None
    except Exception as e:
        print(f"❌ Unknown error: {e}")
        return None


# Usage
shops = safe_api_call(search_shops, "food delivery", "Zhongguancun", user_token)
```

def get_order_with_retry(order_id: str, user_token: str, max_retries: int = 3):
    """
    Query order with retry mechanism
    """
    for attempt in range(max_retries):
        try:
            url = f"{base_url}/api/v1/gateway/orders/{order_id}"
            headers = {
                "Authorization": f"Bearer {api_key}",
                "X-User-Token": user_token
            }
            response = requests.get(url, headers=headers, timeout=5)
            response.raise_for_status()
            return response.json()

        except requests.exceptions.Timeout:
            if attempt < max_retries - 1:
                print(f"Query timeout, retrying... ({attempt + 1}/{max_retries})")
            else:
                print("❌ Cannot get order info, retry later")
                return None

## Complete Example Code

A complete local life Agent example:

```python
import os
import requests
from typing import List, Dict, Optional

class LocalLifeAgent:
    def __init__(self, api_key: str = None):
        self.api_key = api_key or os.getenv("CLAWDOT_API_KEY")
        self.base_url = "https://api.clawdot.com"
        self.user_token = None
        self.user_phone = None

    def bind_user(self, phone: str) -> bool:
        """User binding process"""
        try:
            # Request code
            request_id = self._request_verification(phone)
            # Input code
            code = input("Enter verification code: ")
            # Verify and get token
            self.user_token = self._verify_code(request_id, code)
            self.user_phone = phone
            return True
        except Exception as e:
            print(f"Binding failed: {e}")
            return False

    def search_and_order(self, query: str, location: str) -> Optional[Dict]:
        """Search shops and order"""
        if not self.user_token:
            print("Please bind user account first")
            return None

        # Search shops
        shops = self._search_shops(query, location)
        if not shops:
            print("No matching shops found")
            return None

        # Show results
        for i, shop in enumerate(shops[:5], 1):
            print(f"{i}. {shop['name']} - Rating {shop['rating']}⭐")

        # User selects
        choice = int(input("Select shop (1-5): ")) - 1
        shop = shops[choice]

        # Get menu
        shop_detail = self._get_shop_detail(shop['id'])
        print(f"\n{shop_detail['name']} Menu:")

        # Build order
        items = []
        for category in shop_detail['categories']:
            print(f"\n【{category['name']}】")
            for item in category['items']:
                print(f"  {item['name']} - ¥{item['price']}")

        # Create order
        order = self._create_order(shop['id'], items, location)
        return order

    def _search_shops(self, query: str, location: str) -> List[Dict]:
        """Search shops"""
        url = f"{self.base_url}/api/v1/gateway/shops/search"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token
        }
        response = requests.post(url, json={"query": query, "location": location}, headers=headers)
        return response.json().get("shops", [])

    def _get_shop_detail(self, shop_id: str) -> Dict:
        """Get shop details"""
        url = f"{self.base_url}/api/v1/gateway/shops/{shop_id}"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token
        }
        response = requests.get(url, headers=headers)
        return response.json()

    def _create_order(self, shop_id: str, items: List[Dict], location: str) -> Dict:
        """Create order"""
        url = f"{self.base_url}/api/v1/gateway/orders"
        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "X-User-Token": self.user_token,
            "Content-Type": "application/json"
        }
        payload = {
            "shop_id": shop_id,
            "items": items,
            "delivery_address": location,
            "contact_phone": self.user_phone
        }
        response = requests.post(url, json=payload, headers=headers)
        return response.json()

    def _request_verification(self, phone: str) -> str:
        """Request SMS verification code"""
        url = f"{self.base_url}/api/v1/auth/sms/request"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        response = requests.post(url, json={"phone": phone}, headers=headers)
        return response.json()["request_id"]

    def _verify_code(self, request_id: str, code: str) -> str:
        """Verify SMS code"""
        url = f"{self.base_url}/api/v1/auth/sms/verify"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        response = requests.post(
            url,
            json={"request_id": request_id, "code": code},
            headers=headers
        )
        return response.json()["user_token"]


# Usage example
if __name__ == "__main__":
    agent = LocalLifeAgent()

    # Bind user
    if agent.bind_user("+86 138 0000 0000"):
        # Search and order
        order = agent.search_and_order("food delivery", "No. 1 Zhongguancun Avenue")
        if order:
            print(f"\nOrder created: {order['order_id']}")
```
